import { NextResponse } from "next/server";
import { businessSchema } from "@/lib/validators/business";

interface Params {
  params: { id: string };
}

export async function GET(_: Request, { params }: Params) {
  return NextResponse.json({
    data: { id: params.id },
    message: "Fiche établissement (stub)",
  });
}

export async function PATCH(request: Request, { params }: Params) {
  const body = await request.json().catch(() => null);
  const parsed = businessSchema.partial().safeParse(body);

  if (!parsed.success) {
    return NextResponse.json({ errors: parsed.error.flatten().fieldErrors }, { status: 400 });
  }

  return NextResponse.json(
    { data: { id: params.id, ...parsed.data }, message: "Mise à jour à implémenter avec Prisma" },
    { status: 200 },
  );
}
